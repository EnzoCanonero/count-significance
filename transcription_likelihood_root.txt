\[
\ell(s,\theta)=n\ln\nu-\nu-\frac{1}{2}(\tilde{\theta}-\theta)^2+\text{const.}
\]

\textbf{MLEs.} From
\[
\partial_s \ell=\frac{n}{\nu}-1
\]
and
\[
\partial_\theta \ell=\left(\frac{n}{\nu}-1\right)\sigma b+(\tilde{\theta}-\theta):
\]

\[
\bullet\quad \text{Global: } \hat{\nu}=n \to \hat{\theta}=\tilde{\theta},\quad \hat{s}=n-b_0 e^{\sigma\hat{\theta}}.
\]

\[
\bullet\quad \text{Constrained } \hat{\hat{\theta}}(s) \text{ solves }
\left(\frac{n}{\hat{\hat{\nu}}}-1\right)\sigma\hat{\hat{b}}
=
\hat{\hat{\theta}}-\tilde{\theta}
\text{ (transcendental), with }
\hat{\hat{\nu}}=s+\hat{\hat{b}},\quad
\hat{\hat{b}}=b_0 e^{\sigma\hat{\hat{\theta}}}.
\]

\textbf{Likelihood root.}

\[
r_s
=
\operatorname{sign}(\hat{s}-s)
\sqrt{
2\left[
n\ln\frac{n}{\hat{\hat{\nu}}}
-
(n-\hat{\hat{\nu}})
+
\frac{1}{2}(\hat{\hat{\theta}}-\tilde{\theta})^2
\right]
}.
\]

\[
q_s
=
\sqrt{
\frac{n}{j_{\theta\theta}(s,\hat{\hat{\theta}})}
}
\left[
\ln\frac{n}{\hat{\hat{\nu}}}
+
\frac{\sigma^2\hat{\hat{b}}^2(n-\hat{\hat{\nu}})}{\hat{\hat{\nu}}^2}
\right]
\]
